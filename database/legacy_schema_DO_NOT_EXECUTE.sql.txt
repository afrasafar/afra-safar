-- booking insert is RPC-only: price and owner are always read from the DB.
-- The public function is SECURITY DEFINER deliberately to perform the trusted pricing insert;
-- it has no anonymous EXECUTE, checks auth.uid, restricts fields and sets empty search_path.
create or replace function public.afra_request_booking(
 p_property_id uuid,p_check_in date,p_check_out date,p_guests integer,p_note text default null
) returns uuid language plpgsql security definer set search_path='' as $$
declare v_user uuid := (select auth.uid()); v_property public.properties%rowtype;
 v_amount numeric;v_id uuid;v_max integer;v_base numeric;v_specs public.afra_property_specs%rowtype;v_discount numeric;
begin
 if v_user is null then raise exception 'ورود به حساب الزامی است';end if;
 if p_check_in < current_date or p_check_out<=p_check_in or (p_check_out-p_check_in)>45 then raise exception 'بازه تاریخ نامعتبر است';end if;
 if p_guests<1 or p_guests>50 then raise exception 'تعداد نفرات نامعتبر است';end if;
 select * into v_property from public.properties where id=p_property_id and is_active=true;
 if not found then raise exception 'اقامتگاه فعال نیست';end if;
 v_max := coalesce(v_property.capacity,0);v_base:=coalesce(v_property.price,0);
 select * into v_specs from public.afra_property_specs where property_id=p_property_id;
 if v_max<p_guests or v_base<=0 then raise exception 'ظرفیت یا قیمت قابل تأیید نیست';end if;
 if v_property.owner_id is null or v_property.owner_id=v_user then raise exception 'میزبان معتبر نیست یا رزرو اقامتگاه شخصی مجاز نیست';end if;
 if exists(select 1 from public.afra_property_blocks b where b.property_id=p_property_id
  and daterange(b.start_date,b.end_date,'[)') && daterange(p_check_in,p_check_out,'[)'))
 then raise exception 'بخشی از این تاریخ مسدود است';end if;
 if exists(select 1 from public.afra_booking_orders b where b.property_id=p_property_id
  and b.status in ('confirmed','completed') and b.payment_status <> 'refunded'
  and daterange(b.check_in,b.check_out,'[)') && daterange(p_check_in,p_check_out,'[)'))
 then raise exception 'این تاریخ قبلاً رزرو شده است';end if;
 -- Calendar pricing: explicit nightly override > Iranian weekend (Thursday/Friday) > base rate.
 select coalesce(sum(coalesce(r.night_price,
 case when extract(dow from nights.day)=4 or extract(dow from nights.day)=5
 then coalesce(v_specs.weekend_price,v_base) else v_base end)),0)
 into v_amount
 from generate_series(p_check_in,p_check_out-1,interval '1 day') as nights(day)
 left join public.afra_special_rates r on r.property_id=p_property_id and r.stay_date=nights.day::date;
 -- Extra guest charge per night above standard capacity, if the host defined it.
 if coalesce(v_specs.standard_capacity,0)>0 and p_guests>v_specs.standard_capacity then
  v_amount:=v_amount + (p_guests-v_specs.standard_capacity) * (p_check_out-p_check_in) * coalesce(v_specs.extra_guest_price,0);
 end if;
 -- Never stack two percentage offers without explicit commercial rules.
 v_discount:=greatest(coalesce(v_specs.discount_percent,0),
 case when p_check_out-p_check_in>=7 then coalesce(v_specs.long_stay_discount_percent,0) else 0 end);
 v_amount:=round(v_amount*(1-v_discount/100));
 insert into public.afra_booking_orders(property_id,guest_id,host_id,check_in,check_out,guest_count,
 guest_note,nightly_price,total_amount)
 values (p_property_id,v_user,v_property.owner_id,p_check_in,p_check_out,p_guests,
 left(coalesce(p_note,''),1000),v_base,v_amount) returning id into v_id;
 return v_id;
end;$$;

create or replace function public.afra_host_booking_decision(p_booking_id uuid,p_approve boolean)
returns text language plpgsql security definer set search_path='' as $$
declare v_user uuid:=(select auth.uid());v_row public.afra_booking_orders%rowtype;
begin
 if v_user is null then raise exception 'ورود الزامی است';end if;
 select * into v_row from public.afra_booking_orders where id=p_booking_id for update;
 if not found or v_row.host_id<>v_user or v_row.status<>'pending' then raise exception 'دسترسی یا وضعیت رزرو نامعتبر است';end if;
 update public.afra_booking_orders set status=case when p_approve then 'confirmed' else 'rejected' end,
 payment_deadline_at=case when p_approve then now()+interval '20 minutes' else null end,
 updated_at=now() where id=p_booking_id;
 return case when p_approve then 'confirmed' else 'rejected' end;
end;$$;
revoke all on function public.afra_host_booking_decision(uuid,boolean) from public,anon;
grant execute on function public.afra_host_booking_decision(uuid,boolean) to authenticated;

create or replace function public.afra_guest_cancel_booking(p_booking_id uuid,p_reason text default null)
returns text language plpgsql security definer set search_path='' as $$
declare v_user uuid:=(select auth.uid());v_row public.afra_booking_orders%rowtype;v_status text;
begin
 if v_user is null then raise exception 'ورود الزامی است';end if;
 select * into v_row from public.afra_booking_orders where id=p_booking_id for update;
 if not found or v_row.guest_id<>v_user or v_row.status in ('cancelled','rejected','completed') then raise exception 'درخواست لغو معتبر نیست';end if;
 v_status:=case when v_row.payment_status='paid' then 'cancel_requested' else 'cancelled' end;
 update public.afra_booking_orders set status=v_status,cancellation_reason=left(coalesce(p_reason,''),500),updated_at=now() where id=p_booking_id;
 return v_status;
end;$$;
revoke all on function public.afra_guest_cancel_booking(uuid,text) from public,anon;
grant execute on function public.afra_guest_cancel_booking(uuid,text) to authenticated;

-- The host may reply to approved reviews for their own property.
-- The write is RPC-only; raw UPDATE stays admin/moderator-controlled.
create or replace function public.afra_host_review_reply(p_review_id uuid,p_reply text)
returns boolean language plpgsql security definer set search_path='' as $$
declare v_user uuid:=(select auth.uid());v_review public.afra_reviews%rowtype;
begin
 if v_user is null or length(trim(coalesce(p_reply,'')))<2 or length(p_reply)>1500 then
  raise exception 'متن پاسخ یا هویت نامعتبر است';end if;
 select * into v_review from public.afra_reviews where id=p_review_id for update;
 if not found or v_review.status<>'approved' or not exists(
  select 1 from public.properties p where p.id=v_review.property_id and p.owner_id=v_user
 ) then raise exception 'مجوز پاسخ برای این نظر وجود ندارد';end if;
 update public.afra_reviews set host_reply=trim(p_reply) where id=p_review_id;
 return true;
end;$$;
revoke all on function public.afra_host_review_reply(uuid,text) from public,anon;
grant execute on function public.afra_host_review_reply(uuid,text) to authenticated;
